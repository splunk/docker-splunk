```sh
ensure_syft() {
  ci_bin_dir="$1"
  syft_version="$2"

  require_nonempty "${syft_version}" "SYFT_VERSION"

  mkdir -p "${ci_bin_dir}"
  PATH="${ci_bin_dir}:${PATH}"
  export PATH

  if [ -x "${ci_bin_dir}/syft" ]; then
    return 0
  fi

  os_name="$(uname -s | tr '[:upper:]' '[:lower:]')"
  arch_name="$(uname -m)"
  case "${arch_name}" in
    x86_64|amd64)
      arch_name="amd64"
      ;;
    aarch64|arm64)
      arch_name="arm64"
      ;;
  esac

  # Syft release tarball contains the binary at the root of the archive.
  curl -fsSL \
    "https://github.com/anchore/syft/releases/download/v${syft_version}/syft_${syft_version}_${os_name}_${arch_name}.tar.gz" \
    | tar -xzC "${ci_bin_dir}" syft
  chmod 0755 "${ci_bin_dir}/syft"
}
```

```sh
ensure_crane() {
  ci_bin_dir="$1"
  crane_version="$2"

  require_nonempty "${crane_version}" "CRANE_VERSION"

  mkdir -p "${ci_bin_dir}"
  PATH="${ci_bin_dir}:${PATH}"
  export PATH

  if [ -x "${ci_bin_dir}/crane" ]; then
    return 0
  fi

  os_name="$(uname -s | tr '[:upper:]' '[:lower:]')"
  arch_name="$(uname -m)"
  case "${arch_name}" in
    x86_64|amd64)
      arch_name="amd64"
      ;;
    aarch64|arm64)
      arch_name="arm64"
      ;;
  esac

  # crane release tarball: go-containerregistry_Linux_x86_64.tar.gz at root.
  # OS name: crane uses 'Linux'/'Darwin' (title-case), not lowercase like syft.
  os_title="$(uname -s)"
  case "${arch_name}" in
    amd64)
      crane_arch="x86_64"
      ;;
    arm64)
      crane_arch="arm64"
      ;;
    *)
      crane_arch="${arch_name}"
      ;;
  esac

  curl -fsSL \
    "https://github.com/google/go-containerregistry/releases/download/v${crane_version}/go-containerregistry_${os_title}_${crane_arch}.tar.gz" \
    | tar -xzC "${ci_bin_dir}" crane
  chmod 0755 "${ci_bin_dir}/crane"
}
```

- v0.20.2 asset: `go-containerregistry_Linux_x86_64.tar.gz` (OS is title-case, arch is x86_64 not amd64)
- Archive contains `crane` binary at root — extract it directly with `tar -xzC "${ci_bin_dir}" crane`

The script is POSIX sh (`#!/bin/sh`), follows the generate-release-sbom.sh structure, and uses `jq` (D-05) and `crane` (D-06) to produce CycloneDX JSON (D-07) and human-readable text BOM (D-07).

```sh
#!/bin/sh
set -eu

# Runtime contract
# - Purpose: generate a custom Bill of Materials listing all managed operator images with digests.
# - Inputs: .env for image names/tool versions; release-image-contract.env from publish-release-images.
# - Outputs: CycloneDX JSON BOM, human-readable text BOM, and signing summary under ci-output/.
# - Guardrails: allow_failure initially; crane digest validates image accessibility.

. "${CI_PROJECT_DIR}/gitlab-ci/lib/pipeline-common.sh"

context_file="ci-output/${WORKFLOW_SLUG}-runtime-context.txt"
output_dir="ci-output/${WORKFLOW_SLUG}-output"
summary_file="${output_dir}/summary.txt"
contract_file="${RELEASE_IMAGE_CONTRACT_FILE:-ci-output/publish-release-images-output/release-image-contract.env}"

mkdir -p "ci-output" "${output_dir}"
: > "${context_file}"

load_repo_dotenv "${CI_PROJECT_DIR}/.env"
require_file "${contract_file}" "release image contract"
load_optional_env_file "${contract_file}"
resolve_release_version "${CI_PROJECT_DIR}/Makefile"

# Resolve variables from contract and dotenv
release_version="${RELEASE_VERSION:-${RESOLVED_RELEASE_VERSION}}"
release_image="${RELEASE_IMAGE:-}"
release_distroless_image="${RELEASE_DISTROLESS_IMAGE:-}"
crane_version="${CRANE_VERSION:-}"

require_nonempty "${release_version}" "RELEASE_VERSION"
require_nonempty "${release_image}" "RELEASE_IMAGE from contract"
require_nonempty "${release_distroless_image}" "RELEASE_DISTROLESS_IMAGE from contract"
require_nonempty "${crane_version}" "CRANE_VERSION from .env"

append_context "${context_file}" "release_version" "${release_version}"
append_context "${context_file}" "release_image" "${release_image}"
append_context "${context_file}" "release_distroless_image" "${release_distroless_image}"
append_context "${context_file}" "crane_version" "${crane_version}"

# Install tools
ci_bin_dir="${CI_PROJECT_DIR}/ci-bin"
ensure_ci_bin_path "${ci_bin_dir}"
ensure_crane "${ci_bin_dir}" "${crane_version}"
ensure_jq

# Authenticate to registry for crane digest calls
registry_host="$(registry_host_from_image_ref "${release_image}")"
registry_username="$(first_nonempty "${PIPELINE_DOCKER_USERNAME:-}" "")"
registry_password="$(first_nonempty "${PIPELINE_DOCKER_PASSWORD:-}" "")"
docker_login_registry "${registry_host}" "${registry_username}" "${registry_password}"

# --- Resolve digests via crane ---
# Standard image: multi-arch manifest (linux/amd64 + linux/arm64)
echo "Resolving digest for standard image: ${release_image}"
standard_manifest_digest="$(crane digest "${release_image}")"
standard_amd64_digest="$(crane digest --platform linux/amd64 "${release_image}")"
standard_arm64_digest="$(crane digest --platform linux/arm64 "${release_image}")"

echo "Resolving digest for distroless image: ${release_distroless_image}"
distroless_manifest_digest="$(crane digest "${release_distroless_image}")"
distroless_amd64_digest="$(crane digest --platform linux/amd64 "${release_distroless_image}")"

require_nonempty "${standard_manifest_digest}" "standard image manifest digest"
require_nonempty "${distroless_manifest_digest}" "distroless image manifest digest"

append_context "${context_file}" "standard_manifest_digest" "${standard_manifest_digest}"
append_context "${context_file}" "standard_amd64_digest" "${standard_amd64_digest}"
append_context "${context_file}" "standard_arm64_digest" "${standard_arm64_digest}"
append_context "${context_file}" "distroless_manifest_digest" "${distroless_manifest_digest}"
append_context "${context_file}" "distroless_amd64_digest" "${distroless_amd64_digest}"

# Extract image name and tag components
image_repo="${RELEASE_IMAGE_REPOSITORY:-splunk/splunk-operator}"
image_tag_standard="${release_version}"
image_tag_distroless="${release_version}-distroless"

# Strip sha256: prefix from digests for purl hash field
standard_manifest_sha="$(printf '%s' "${standard_manifest_digest}" | sed 's/sha256://')"
distroless_manifest_sha="$(printf '%s' "${distroless_manifest_digest}" | sed 's/sha256://')"

# --- CycloneDX 1.5 JSON BOM (D-05 jq, D-07 CycloneDX) ---
bom_json="${output_dir}/bom-v${release_version}.json"
bom_timestamp="$(date -u +"%Y-%m-%dT%H:%M:%SZ")"

jq -n \
  --arg timestamp "${bom_timestamp}" \
  --arg version "${release_version}" \
  --arg image_repo "${image_repo}" \
  --arg tag_standard "${image_tag_standard}" \
  --arg tag_distroless "${image_tag_distroless}" \
  --arg digest_standard "${standard_manifest_digest}" \
  --arg digest_distroless "${distroless_manifest_digest}" \
  --arg sha_standard "${standard_manifest_sha}" \
  --arg sha_distroless "${distroless_manifest_sha}" \
  '{
    "bomFormat": "CycloneDX",
    "specVersion": "1.5",
    "version": 1,
    "metadata": {
      "timestamp": $timestamp,
      "tools": [
        {
          "vendor": "google",
          "name": "crane",
          "version": "'"${crane_version}"'"
        }
      ],
      "component": {
        "type": "application",
        "name": "splunk-operator",
        "version": $version
      }
    },
    "components": [
      {
        "type": "container",
        "name": ($image_repo + ":" + $tag_standard),
        "version": $tag_standard,
        "purl": ("pkg:oci/" + $image_repo + ":" + $tag_standard + "@" + $digest_standard + "?repository_url=registry-1.docker.io"),
        "hashes": [
          {
            "alg": "SHA-256",
            "content": $sha_standard
          }
        ],
        "properties": [
          {"name": "splunk:variant", "value": "standard"},
          {"name": "splunk:platforms", "value": "linux/amd64,linux/arm64"}
        ]
      },
      {
        "type": "container",
        "name": ($image_repo + ":" + $tag_distroless),
        "version": $tag_distroless,
        "purl": ("pkg:oci/" + $image_repo + ":" + $tag_distroless + "@" + $digest_distroless + "?repository_url=registry-1.docker.io"),
        "hashes": [
          {
            "alg": "SHA-256",
            "content": $sha_distroless
          }
        ],
        "properties": [
          {"name": "splunk:variant", "value": "distroless"},
          {"name": "splunk:platforms", "value": "linux/amd64"}
        ]
      }
    ]
  }' > "${bom_json}"

echo "BOM JSON written: ${bom_json}"

# --- Human-readable text BOM (D-07 human-readable) ---
bom_txt="${output_dir}/bom-v${release_version}.txt"

{
  printf "# Splunk Operator %s - Bill of Materials\n" "${release_version}"
  printf "# Generated: %s\n" "${bom_timestamp}"
  printf "#\n"
  printf "%-60s | %-40s | %-80s | %-20s\n" "IMAGE_NAME" "TAG" "DIGEST" "PLATFORM"
  printf "%s\n" "$(printf '%0.s-' $(seq 1 210))"
  printf "%-60s | %-40s | %-80s | %-20s\n" "${image_repo}" "${image_tag_standard}" "${standard_manifest_digest}" "linux/amd64,linux/arm64"
  printf "%-60s | %-40s | %-80s | %-20s\n" "${image_repo}" "${image_tag_standard}" "${standard_amd64_digest}" "linux/amd64"
  printf "%-60s | %-40s | %-80s | %-20s\n" "${image_repo}" "${image_tag_standard}" "${standard_arm64_digest}" "linux/arm64"
  printf "%-60s | %-40s | %-80s | %-20s\n" "${image_repo}" "${image_tag_distroless}" "${distroless_manifest_digest}" "linux/amd64"
  printf "%-60s | %-40s | %-80s | %-20s\n" "${image_repo}" "${image_tag_distroless}" "${distroless_amd64_digest}" "linux/amd64"
} > "${bom_txt}"

echo "BOM text written: ${bom_txt}"

# --- Signing summary (D-10 signing-summary.md) ---
signing_summary="${output_dir}/signing-summary.md"

cat > "${signing_summary}" <<SIGNING_EOF
# Splunk Operator ${release_version} - Supply Chain Verification Summary

Generated: ${bom_timestamp}

## Image Verification (Cosign)

> Note: Cosign signing is planned for a future release. The commands below will
> work once signing is enabled. See docs/supply-chain-verification.md for the
> public key location.

\`\`\`sh
# Download the Splunk Operator public key
curl -fsSL https://github.com/splunk/splunk-operator/raw/main/cosign.pub -o cosign.pub

# Verify standard operator image
cosign verify --key cosign.pub ${release_image}@${standard_manifest_digest}

# Verify distroless operator image
cosign verify --key cosign.pub ${release_distroless_image}@${distroless_manifest_digest}
\`\`\`

## Image Digests

| Image | Tag | Manifest Digest | Platforms |
|-------|-----|-----------------|-----------|
| splunk-operator | ${image_tag_standard} | \`${standard_manifest_digest}\` | linux/amd64, linux/arm64 |
| splunk-operator | ${image_tag_distroless} | \`${distroless_manifest_digest}\` | linux/amd64 |

## SBOM Download & Inspection

\`\`\`sh
# Download SBOMs from the GitLab release page
# (URLs use version-specific paths; replace VERSION as needed)
curl -fsSL https://cd.splunkdev.com/sok/splunk-operator/-/releases/v${release_version}/downloads/sbom/sbom-operator-v${release_version}.cyclonedx.json -o sbom-standard.json
curl -fsSL https://cd.splunkdev.com/sok/splunk-operator/-/releases/v${release_version}/downloads/sbom/sbom-operator-v${release_version}-distroless.cyclonedx.json -o sbom-distroless.json

# Scan for vulnerabilities with grype
grype sbom:sbom-standard.json

# Scan with trivy
trivy sbom sbom-standard.json
\`\`\`

## BOM Reference

- CycloneDX BOM (JSON): bom-v${release_version}.json
- Human-readable BOM (text): bom-v${release_version}.txt

See docs/supply-chain-verification.md for complete verification instructions.
SIGNING_EOF

echo "Signing summary written: ${signing_summary}"

# --- Summary ---
component_count="$(jq '.components | length' "${bom_json}")"

cat > "${summary_file}" <<EOF
Generated Bill of Materials for operator release images.

- release_version: ${release_version}
- crane_version: ${crane_version}
- standard_image: ${release_image}
- standard_manifest_digest: ${standard_manifest_digest}
- distroless_image: ${release_distroless_image}
- distroless_manifest_digest: ${distroless_manifest_digest}
- bom_json: bom-v${release_version}.json (${component_count} components)
- bom_txt: bom-v${release_version}.txt
- signing_summary: signing-summary.md
EOF

echo ""
echo "=== BOM Generation Complete ==="
cat "${summary_file}"
```
